import { CollapsibleSection, EmptyState, GlassCard } from "../shared/ui";

// ── Design tokens for vivah panel ─────────────────────────
const passColor  = "#22D3EE";   // cyan
const warnColor  = "#FCD34D";   // yellow — visible on dark
const failColor  = "#FB7185";   // rose
const neutColor  = "#94A3B8";   // slate
const infoColor  = "#C084FC";   // purple
const goldColor  = "#F59E0B";   // amber

// ── Atoms ─────────────────────────────────────────────────
function Verdict({ text, color }) {
  return (
    <div style={{
      display:"inline-block", padding:"5px 14px",
      borderRadius:"20px", marginBottom:"10px",
      background:`${color}18`, border:`1.5px solid ${color}55`,
      fontSize:"0.82rem", fontWeight:700, color, ...HI,
    }}>{text}</div>
  );
}

function SutraRow({ text, icon = "•", color = "rgba(255,255,255,0.8)" }) {
  return (
    <div style={{
      display:"flex", gap:"8px", alignItems:"flex-start",
      padding:"7px 10px", borderRadius:"8px", marginBottom:"5px",
      background:"rgba(255,255,255,0.05)",
      border:"1px solid rgba(255,255,255,0.1)",
    }}>
      <span style={{ fontSize:"0.85rem", flexShrink:0, marginTop:"1px" }}>{icon}</span>
      <span style={{ ...HI, fontSize:"0.79rem", color, lineHeight:1.5 }}>{text}</span>
    </div>
  );
}

function Section2Col({ title, left, right }) {
  return (
    <div>
      <div style={{ ...HI, fontSize:"0.7rem", color:"rgba(255,255,255,0.4)", marginBottom:"8px" }}>{title}</div>
      <div style={{ display:"grid", gridTemplateColumns:"1fr 1fr", gap:"8px" }}>
        <div>{left}</div>
        <div>{right}</div>
      </div>
    </div>
  );
}

function PlanetChipRow({ label, planets, color }) {
  if (!planets?.length) return null;
  return (
    <div style={{ display:"flex", gap:"6px", alignItems:"center", flexWrap:"wrap", marginBottom:"6px" }}>
      <span style={{ fontSize:"0.7rem", color:"rgba(255,255,255,0.45)", ...HI, minWidth:"80px" }}>{label}:</span>
      {planets.map((p,i) => (
        <span key={i} style={{
          padding:"2px 10px", borderRadius:"8px", fontSize:"0.75rem", fontWeight:700,
          background:`${color}20`, border:`1px solid ${color}50`, color, ...HI,
        }}>{p}</span>
      ))}
    </div>
  );
}

function Divider({ color = "rgba(255,255,255,0.08)" }) {
  return <div style={{ height:"1px", background:color, margin:"10px 0" }} />;
}

          )}

          <Divider />
          <SutraRow icon="📜" text={vd.dwi_vivah?.note} color="rgba(255,255,255,0.5)" />
        </div>
      </CollapsibleSection>

      {/* ─── 12. VAIDHAVYA (Stri kundli) ─────────────────── */}
      {vd.vaidhavya?.applicable !== false && (
        <CollapsibleSection icon="🕯️" title="वैधव्य योग (स्त्री कुंडली)" color={vd.vaidhavya?.has_risk ? failColor : passColor} defaultOpen={false}>
          <div style={{ padding:"4px 0 8px" }}>
            <Verdict text={vd.vaidhavya?.verdict} color={vd.vaidhavya?.has_risk ? failColor : passColor} />

            {vd.vaidhavya?.factors?.length > 0 && (
              <>
                <div style={{ ...HI, fontSize:"0.72rem", color:"rgba(255,255,255,0.45)", marginBottom:"6px" }}>⚠️ वैधव्य के संकेत:</div>
                {vd.vaidhavya.factors.map((f,i) => <SutraRow key={i} icon="⚠️" text={f} color={failColor} />)}
              </>
            )}

            {vd.vaidhavya?.protect?.length > 0 && (
              <>
                <Divider />
                <div style={{ ...HI, fontSize:"0.72rem", color:"rgba(255,255,255,0.45)", marginBottom:"6px" }}>🛡️ सुरक्षा कारक:</div>
                {vd.vaidhavya.protect.map((f,i) => <SutraRow key={i} icon="🛡️" text={f} color={passColor} />)}
              </>
            )}

            <Divider />
            <SutraRow icon="📜" text={vd.vaidhavya?.note_8th} color="rgba(255,255,255,0.5)" />
          </div>
        </CollapsibleSection>
      )}

      {/* ─── 13. VYABHICHAR (Charitra) ─────────────────────── */}
      <CollapsibleSection icon="⚖️" title="चरित्र विश्लेषण" color={vd.vyabhichar?.has_risk ? failColor : neutColor} defaultOpen={false}>
        <div style={{ padding:"4px 0 8px" }}>
          <Verdict text={vd.vyabhichar?.verdict} color={vd.vyabhichar?.has_risk ? failColor : passColor} />

          {vd.vyabhichar?.factors?.length > 0 && (
            <>
              <div style={{ ...HI, fontSize:"0.72rem", color:"rgba(255,255,255,0.45)", marginBottom:"6px" }}>⚠️ चरित्र पर संकेत:</div>
              {vd.vyabhichar.factors.map((f,i) => <SutraRow key={i} icon="⚠️" text={f} color={warnColor} />)}
            </>
          )}

          {vd.vyabhichar?.protect_factors?.length > 0 && (
            <>
              <Divider />
              <div style={{ ...HI, fontSize:"0.72rem", color:"rgba(255,255,255,0.45)", marginBottom:"6px" }}>✅ नैतिकता के रक्षक:</div>
              {vd.vyabhichar.protect_factors.map((f,i) => <SutraRow key={i} icon="✅" text={f} color={passColor} />)}
            </>
          )}

          <Divider />
          <SutraRow icon="📌" text={vd.vyabhichar?.important_note || "ये संकेत हैं — दशा + परिस्थिति + इच्छाशक्ति भी भूमिका निभाती है।"} color={goldColor} />
        </div>
      </CollapsibleSection>

      {/* ─── NOTE ─────────────────────────────────────────── */}
      <GlassCard className="p-3" style={{ background:"rgba(255,255,255,0.03)" }}>
        <div style={{ ...HI, fontSize:"0.72rem", color:"rgba(255,255,255,0.4)", lineHeight:1.6 }}>
          📌 <strong style={{ color:"rgba(255,255,255,0.6)" }}>नोट:</strong> ये सभी संकेत हैं — अंतिम निर्णय आप स्वयं लें।
          दशा + गोचर का मिलान भी देखें। गुरु की दृष्टि अशुभ योगों को कम करती है।
        </div>
      </GlassCard>

    </div>
  );
}
